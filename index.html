function ProfileCard({ name, role }) {
  const [following, setFollowing] = useState(false);

  return (
    <div className="card">
      <h4>{name}</h4>
      <p>{role}</p>
      <button onClick={() => setFollowing(!following)}>
        {following ? 'Following ✓' : 'Follow'}
      </button>
    </div>
  );
}

Frontend structure

Know where every file goes

Beginners often get stuck not on React itself, but on how to organize a project. This kit shows the layout real teams use, so your projects look professional from the start.

  • components/

    Small reusable pieces like buttons, cards and modals.

  • pages/

    Full screens such as Home, About and Dashboard.

  • hooks/

    Reusable logic like useFetch or useToggle.

  • services/

    Everything that talks to an API or server.
